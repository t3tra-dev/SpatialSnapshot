cmake_minimum_required(VERSION 3.20)
project(SpatialSnapshotABIConsumer LANGUAGES CXX)
find_package(SpatialSnapshot 1.0 CONFIG REQUIRED)
add_executable(ss_abi_consumer consumer.cpp)
target_compile_features(ss_abi_consumer PRIVATE cxx_std_17)
if(NOT MSVC)
    target_compile_options(ss_abi_consumer PRIVATE -Wall -Wextra -Wpedantic)
endif()
target_link_libraries(ss_abi_consumer PRIVATE SpatialSnapshot::C)
